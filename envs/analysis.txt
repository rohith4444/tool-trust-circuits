# "analysis" environment -- TransformerLens + EAP-IG. Runs on the pod (GPU).
#
#   python3 -m venv .venv-analysis              (or: uv venv .venv-analysis)
#   .venv-analysis/bin/pip install -r envs/analysis.txt -e .
#
# On the pod put the venv on the container disk like VENV_GEN (see scripts/pod_env.sh).
#
# transformer-lens 2.18.0 is the last 2.x release. HookedTransformer is its primary
# API; 3.x deprecates it for TransformerBridge and needs transformers>=5.9.
# It hardcodes n_ctx=2048 for Qwen2 -- tooltrust/models/load.py overrides n_ctx.
# It pins numpy<2 only on Python <=3.12. numpy 1.x has no 3.13 wheels, so an
# unconditional numpy<2 makes pip try a source build and abort (Colab is 3.13).
transformer-lens==2.18.0
transformers>=4.57,<5
torch>=2.6
numpy<2; python_version < "3.13"

# transformer_lens imports datasets, which is otherwise unpinned and resolved to 2.14.4
# on a fresh machine -- that version uses pa.PyExtensionType, removed in pyarrow 15, so
# `import transformer_lens` dies before anything runs. Upgrading it unpinned then pulls
# huggingface_hub 1.x, which transformers<5 refuses. Both bounds are needed.
datasets>=2.16,<4
huggingface_hub>=0.34,<1.0
pandas>=2,<3

# The TIR sandbox and grader, for scripts/run_regenerate.py: it re-runs the tool loop, so
# it executes model-written Python and grades the result the same way generation did.
Pebble
timeout-decorator
sympy
latex2sympy2
word2number

# EAP-IG is not on PyPI and the package name is `eap` (an unrelated `eap` exists on
# PyPI -- do not `pip install eap`). Its scorers hardcode device='cuda', so the EAP
# stage only runs on the pod. Commit pinned at implementation time.
eap @ git+https://github.com/hannamw/EAP-IG.git@e24bafd3d22af2c59ac5a83e0a6581b89b5c05dd
